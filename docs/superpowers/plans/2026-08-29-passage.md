# Passage — plan de mise en œuvre

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construire *Passage*, une application Expo/React Native personnelle qui propose chaque jour cinq phrases françaises à traduire en anglais et les fait corriger par Claude, avec adaptation du niveau aux erreurs passées.

**Architecture:** Cinq couches à dépendances descendantes — `ui/` → `app/` (cas d'usage) → `ai/` et `data/` (bordures réseau et disque) → `core/` (logique pure sans dépendance). Les deux bordures sont derrière des interfaces, ce qui rend chaque cas d'usage testable avec des doublures. La série du lendemain est pré-générée après chaque correction, de sorte que la phase de traduction fonctionne hors-ligne.

**Tech Stack:** Expo SDK 57, React Native, TypeScript, `expo-router`, `expo-sqlite`, `expo-secure-store`, `expo-notifications`, `react-native-svg`, `@anthropic-ai/sdk` 0.122, `zod` 4.5, Jest (`jest-expo` + projet `node`), `@testing-library/react-native`, `node:sqlite` pour les tests de persistance.

**Spec:** `docs/superpowers/specs/2026-08-29-passage-design.md`

## Global Constraints

- Modèle Claude : `claude-opus-5` exactement. Jamais de suffixe de date.
- Pensée adaptative : `thinking: { type: 'adaptive' }`. Ne **jamais** passer `budget_tokens` — rejeté par un 400 sur ce modèle.
- `output_config.effort` : `'low'` pour la génération, `'high'` pour la correction.
- `max_tokens: 16000`, appels non-streamés.
- Sorties structurées via `client.messages.parse({ output_config: { format: zodOutputFormat(Schema) } })`. `response.parsed_output` peut être `null` — toujours vérifier.
- La clé API n'est jamais écrite dans le dépôt, jamais journalisée, jamais mise dans un message d'erreur. Elle vit uniquement dans `expo-secure-store`.
- Langue de l'interface : français. Langue cible des exercices : anglais.
- Les erreurs remontées à `ui/` sont d'un type fermé : `invalid_key | rate_limited | offline | bad_response | api_error`.
- Ensemble fermé d'étiquettes d'erreur, dans cet ordre :
  `tense`, `preposition`, `article`, `word_order`, `vocabulary`, `false_friend`, `agreement`, `register`, `spelling`, `idiom`.
- Niveaux, dans cet ordre : `A2`, `B1`, `B2`, `C1`, `C2`. Niveau de base choisi parmi `A2`–`C1`.
- Palette (jetons de thème, valeurs exactes) :
  clair — fond `#FBF9F4`, texte `#1A1815`, secondaire `#6B6459`, accent `#C2703D`, réussite `#4A7C59`, erreur `#B4544A`, bordure `#E5DFD4` ;
  sombre — fond `#121110`, texte `#F0EBE3`, secondaire `#9A9186`, accent `#D98A56`, réussite `#6FA37D`, erreur `#D2766B`, bordure `#2A2724`.

const pad = (n: number): string => String(n).padStart(2, '0');

export function localDay(d: Date): string {
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
}

export function addDays(day: string, n: number): string {
  const [y, m, d] = day.split('-').map(Number);
  return localDay(new Date(y, m - 1, d + n));
}

export const LEVELS = ['A2', 'B1', 'B2', 'C1', 'C2'] as const;
export type Level = (typeof LEVELS)[number];

export const BASE_LEVELS: readonly Level[] = ['A2', 'B1', 'B2', 'C1'];

const WINDOW = 5;
const MIN_SERIES = 3;
const UP = 8.5;
const DOWN = 5.5;

export function effectiveLevel(base: Level, recentAverages: number[]): Level {
  const window = recentAverages.slice(-WINDOW);
  if (window.length < MIN_SERIES) return base;

  const mean = window.reduce((a, b) => a + b, 0) / window.length;
  const shift = mean >= UP ? 1 : mean <= DOWN ? -1 : 0;

  const index = LEVELS.indexOf(base) + shift;
  return LEVELS[Math.min(Math.max(index, 0), LEVELS.length - 1)];
}
```

export type DiffOp = { op: 'keep' | 'del' | 'ins'; text: string };

const tokenize = (s: string): string[] => s.match(/[\p{L}\p{N}']+|[^\s\p{L}\p{N}]/gu) ?? [];

const same = (a: string, b: string) => a.toLowerCase() === b.toLowerCase();

export function wordDiff(from: string, to: string): DiffOp[] {
  const a = tokenize(from);
  const b = tokenize(to);

  // Table de plus longue sous-séquence commune.
  const lcs: number[][] = Array.from({ length: a.length + 1 }, () =>
    new Array<number>(b.length + 1).fill(0),
  );
  for (let i = a.length - 1; i >= 0; i--) {
    for (let j = b.length - 1; j >= 0; j--) {
      lcs[i][j] = same(a[i], b[j])
        ? lcs[i + 1][j + 1] + 1
        : Math.max(lcs[i + 1][j], lcs[i][j + 1]);
    }
  }

  const ops: DiffOp[] = [];
  let i = 0;
  let j = 0;
  while (i < a.length && j < b.length) {
    if (same(a[i], b[j])) {
      ops.push({ op: 'keep', text: b[j] });   // on affiche la forme correcte
      i++;
      j++;
    } else if (lcs[i + 1][j] >= lcs[i][j + 1]) {
      ops.push({ op: 'del', text: a[i++] });
    } else {
      ops.push({ op: 'ins', text: b[j++] });
    }
  }
  while (i < a.length) ops.push({ op: 'del', text: a[i++] });
  while (j < b.length) ops.push({ op: 'ins', text: b[j++] });

  return ops;
}
```

export const ERROR_TAGS = [
  'tense', 'preposition', 'article', 'word_order', 'vocabulary',
  'false_friend', 'agreement', 'register', 'spelling', 'idiom',
] as const;

export type ErrorTag = (typeof ERROR_TAGS)[number];

export const TAG_LABELS_FR: Record<ErrorTag, string> = {
  tense: 'Temps',
  preposition: 'Préposition',
  article: 'Article',
  word_order: 'Ordre des mots',
  vocabulary: 'Vocabulaire',
  false_friend: 'Faux-ami',
  agreement: 'Accord',
  register: 'Registre',
  spelling: 'Orthographe',
  idiom: 'Tournure idiomatique',
};

const LOOKBACK = 3;

export function topWeakTags(seriesTags: ErrorTag[][], limit = 3): ErrorTag[] {
  const counts = new Map<ErrorTag, number>();
  for (const tags of seriesTags.slice(-LOOKBACK)) {
    for (const tag of tags) counts.set(tag, (counts.get(tag) ?? 0) + 1);
  }

  return [...counts.entries()]
    .sort((a, b) => b[1] - a[1] || ERROR_TAGS.indexOf(a[0]) - ERROR_TAGS.indexOf(b[0]))
    .slice(0, limit)
    .map(([tag]) => tag);
}
```

import { addDays } from './date';

export function currentStreak(correctedDays: string[], today: string): number {
  const days = new Set(correctedDays);

  // La série reste vivante tant qu'aujourd'hui n'est pas terminé :
  // on part d'aujourd'hui s'il est fait, sinon d'hier.
  let cursor = days.has(today) ? today : addDays(today, -1);
  if (!days.has(cursor)) return 0;

  let count = 0;
  while (days.has(cursor)) {
    count++;
    cursor = addDays(cursor, -1);
  }
  return count;
}
```

import Anthropic from '@anthropic-ai/sdk';

export type AppErrorKind =
  | 'invalid_key' | 'rate_limited' | 'offline' | 'bad_response' | 'api_error';

const MESSAGES: Record<AppErrorKind, string> = {
  invalid_key: "La clé API est refusée. Vérifie-la dans les réglages.",
  rate_limited: 'Trop de requêtes. Réessaie dans un instant.',
  offline: 'Pas de connexion.',
  bad_response: "Réponse illisible du modèle.",
  api_error: "L'API a renvoyé une erreur.",
};

export class AppError extends Error {
  constructor(
    readonly kind: AppErrorKind,
    message?: string,
    readonly status?: number,
  ) {
    super(message ?? MESSAGES[kind]);
    this.name = 'AppError';
  }
}

export function toAppError(e: unknown): AppError {
  if (e instanceof AppError) return e;
  if (e instanceof Anthropic.AuthenticationError) return new AppError('invalid_key', undefined, 401);
  if (e instanceof Anthropic.RateLimitError) return new AppError('rate_limited', undefined, 429);
  if (e instanceof Anthropic.APIConnectionError) return new AppError('offline');
  if (e instanceof Anthropic.APIError) return new AppError('api_error', undefined, e.status);
  // Message générique : le message d'origine peut contenir la clé.
  return new AppError('api_error');
}

import type { Level } from '../core/levels';
import type { ErrorTag } from '../core/errorTags';

export function generationUserMessage(input: {
  level: Level;
  weakTags: ErrorTag[];
  recentSources: string[];
}): string {
  const parts = [`Niveau : ${input.level}`];
  parts.push(
    input.weakTags.length
      ? `Points faibles à retravailler, par ordre de priorité : ${input.weakTags.join(', ')}.`
      : `Aucun point faible identifié pour l'instant : balaie large.`,
  );
  if (input.recentSources.length) {
    parts.push(
      `Phrases déjà proposées ces derniers jours — ne les reprends pas et évite les sujets trop proches :\n` +
        input.recentSources.map((s) => `- ${s}`).join('\n'),
    );
  }
  return parts.join('\n\n');
}

export function correctionUserMessage(input: {
  level: Level;
  items: { position: number; source_fr: string; reference_en: string; user_en: string }[];
}): string {
  const blocks = input.items.map(
    (it) =>
      `Phrase ${it.position}\n` +
      `Source : ${it.source_fr}\n` +
      `Référence : ${it.reference_en}\n` +
      `Réponse de l'apprenant : ${it.user_en || '(vide)'}`,
  );
  return `Niveau visé : ${input.level}\n\n${blocks.join('\n\n')}`;
}

import Anthropic from '@anthropic-ai/sdk';
import { zodOutputFormat } from '@anthropic-ai/sdk/helpers/zod';
import { AppError, toAppError } from './errors';
import { CorrectionSchema, GeneratedSeriesSchema } from './schemas';
import {
  CORRECTION_SYSTEM, GENERATION_SYSTEM,
  correctionUserMessage, generationUserMessage,
} from './prompts';
import type { Level } from '../core/levels';
import type { ErrorTag } from '../core/errorTags';

const MODEL = 'claude-opus-5';
const MAX_TOKENS = 16000;

async function parseWithRetry<T>(run: () => Promise<{ parsed_output: T | null }>): Promise<T> {
  for (let attempt = 0; attempt < 2; attempt++) {
    let response;
    try {
      response = await run();
    } catch (e) {
      throw toAppError(e);
    }
    if (response.parsed_output) return response.parsed_output;
  }
  throw new AppError('bad_response');
}

export interface AiClient { /* signature de la section Interfaces */ }

export function makeAiClient(apiKey: string): AiClient {
  const client = new Anthropic({ apiKey });

  return {
    async generateSeries(input) {
      const parsed = await parseWithRetry(() =>
        client.messages.parse({
          model: MODEL,
          max_tokens: MAX_TOKENS,
          thinking: { type: 'adaptive' },
          output_config: { effort: 'low', format: zodOutputFormat(GeneratedSeriesSchema) },
          system: [{ type: 'text', text: GENERATION_SYSTEM, cache_control: { type: 'ephemeral' } }],
          messages: [{ role: 'user', content: generationUserMessage(input) }],
        }),
      );
      return parsed.sentences;
    },

    async correctSeries(input) {
      const parsed = await parseWithRetry(() =>
        client.messages.parse({
          model: MODEL,
          max_tokens: MAX_TOKENS,
          thinking: { type: 'adaptive' },
          output_config: { effort: 'high', format: zodOutputFormat(CorrectionSchema) },
          system: [{ type: 'text', text: CORRECTION_SYSTEM, cache_control: { type: 'ephemeral' } }],
          messages: [{ role: 'user', content: correctionUserMessage(input) }],
        }),
      );
      return { ...parsed, items: [...parsed.items].sort((a, b) => a.position - b.position) };
    },
  };
}
```

export function resolveLevel(deps: Deps): Level {
  const base = (deps.settings.get('base_level') ?? 'B1') as Level;
  return effectiveLevel(base, deps.stats.recentAverages(5));
}

export async function getTodaySeries(deps: Deps): Promise<StoredSeries> {
  const day = localDay(deps.now());
  const existing = deps.series.findByDay(day);
  if (existing) return existing;

  const level = resolveLevel(deps);
  const weakTags = topWeakTags(deps.stats.recentTagsBySeries(3));
  const sentences = await deps.ai.generateSeries({
    level,
    weakTags,
    recentSources: deps.series.recentSources(7),
  });
  return deps.series.insert(day, level, sentences);
}
```

```ts
export async function submitAnswers(
  deps: Deps,
  seriesId: number,
  answers: { position: number; user_en: string }[],
): Promise<StoredSeries> {
  deps.series.saveAnswers(seriesId, answers);
  deps.series.setStatus(seriesId, 'in_progress');

  const series = deps.series.findByDay(localDay(deps.now()));
  if (!series) throw new AppError('bad_response', 'Série introuvable.');

  try {
    const correction = await deps.ai.correctSeries({
      level: series.level,
      items: series.sentences.map((s) => ({
        position: s.position,
        source_fr: s.source_fr,
        reference_en: s.reference_en,
        user_en: s.user_en ?? '',
      })),
    });
    deps.series.saveCorrections(seriesId, correction.items);
  } catch (e) {
    deps.series.setStatus(seriesId, 'awaiting_correction');
    throw toAppError(e);
  }

  // Le prefetch ne doit jamais faire échouer la correction.
  void prefetchTomorrow(deps).catch(() => undefined);

  return deps.series.findByDay(series.day)!;
}
```
